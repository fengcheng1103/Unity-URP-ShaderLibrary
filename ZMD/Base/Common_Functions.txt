// 1. 金属度影响漫反射
half OneMinusReflectivityMetallic(half metallic) {
    return 1.0 - metallic * 0.04; // 近似
}

// 2. GGX 分布（Endfield 自定义）
float D_DistributionGGX_Endfield(float NoH, float roughness) {
    float a = roughness * roughness;
    float a2 = a * a;
    float denom = NoH * NoH * (a2 - 1.0) + 1.0;
    return a2 / (PI * denom * denom);
}

// 3. Smith 几何衰减
float G_GeometrySmith_Endfield(float NoV, float NoL, float roughness) {
    float k = (roughness + 1.0) * (roughness + 1.0) / 8.0;
    float G1V = NoV / (NoV * (1.0 - k) + k);
    float G1L = NoL / (NoL * (1.0 - k) + k);
    return G1V * G1L;
}

// 4. 菲涅尔（直接光）
float3 F_FrenelSchlick_Direct_Endfield(float NoV, float3 F0) {
    return F0 + (1.0 - F0) * pow(1.0 - NoV, 5.0);
}

// 6. 采样Ramp图（自定义）
half4 SamplerRampMap(TEXTURE2D_PARAM(tex, sampler), float uv, float strength) {
    half4 ramp = SAMPLE_TEXTURE2D(tex, sampler, float2(uv, 0.5));
    return lerp(half4(1,1,1,1), ramp, strength);
}

// 7. RimLightMask（自定义）
float GetRimLightMask(float offset, float softness, float posX, float3 normal, float3 direction) {
    // 例如基于视角方向和法线的点积
    float mask = dot(normal, direction) + offset;
    mask = smoothstep(0, 1, saturate(mask * softness));
    return mask;
}
